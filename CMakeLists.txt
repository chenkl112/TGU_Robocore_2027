cmake_minimum_required(VERSION 3.16)
project(TGU_Robocore_2027 VERSION 0.1.0 LANGUAGES CXX)

option(ROBOCORE_WITH_VISION "Build OpenVINO detection and replay pipeline" ON)
option(ROBOCORE_WITH_IO "Build serial, gimbal, and camera devices" ON)
option(ROBOCORE_WITH_FOXGLOVE "Build the optional Foxglove SDK adapter" ON)
option(ROBOCORE_BUILD_DEMOS "Build small diagnostic executables" ON)
option(ROBOCORE_BUILD_CAMERA_DEMO "Build the Aravis camera demo" OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

include(cmake/RobocoreTargets.cmake)
# 在顶层声明共享目标，让 find_package 的导入目标对所有测试子目录可见。
set(robocore_components core legacy image)
if(ROBOCORE_WITH_VISION)
    list(APPEND robocore_components vision)
endif()
if(ROBOCORE_WITH_IO)
    list(APPEND robocore_components serial gimbal camera)
endif()
if(ROBOCORE_WITH_FOXGLOVE)
    list(APPEND robocore_components foxglove)
endif()
robocore_add_targets(COMPONENTS ${robocore_components})
add_subdirectory(tools)
add_subdirectory(app)
if(ROBOCORE_WITH_IO)
    add_subdirectory(io)
endif()
add_subdirectory(task)

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests/infra)
    if(ROBOCORE_WITH_IO)
        add_subdirectory(tests/io)
    endif()
    add_subdirectory(tests/auto_aim)
endif()
