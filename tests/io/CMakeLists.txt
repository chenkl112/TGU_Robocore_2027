cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(robocore_io_tests LANGUAGES CXX)
    enable_testing()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The serial lifecycle tests require Linux poll and pseudo-terminals.")
endif()
set(IO_SANITIZER "none" CACHE STRING "none, address (with UBSan), or thread")
set_property(CACHE IO_SANITIZER PROPERTY STRINGS none address thread)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/RobocoreTargets.cmake")
robocore_add_targets(COMPONENTS serial gimbal)

add_executable(serial_tests serial_tests.cpp)
target_link_libraries(serial_tests PRIVATE robocore_serial util)
add_executable(gimbal_lifecycle_tests gimbal_lifecycle_tests.cpp)
target_link_libraries(gimbal_lifecycle_tests PRIVATE robocore_gimbal util)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(target serial_tests gimbal_lifecycle_tests)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endforeach()
endif()

foreach(scenario IN ITEMS read_timeout close_read fragmented subscriptions false_header isolated_ports
        reopen failed_open disconnect write_timeout close_write queued_write callback_close callback_throw
        concurrent guarded_cancel)
    add_test(NAME serial.${scenario} COMMAND serial_tests ${scenario})
    set_tests_properties(serial.${scenario} PROPERTIES TIMEOUT 15)
endforeach()
foreach(scenario IN ITEMS silent_start idle_shutdown feedback_timeout pose_history mode_session
        feedback_alignment reconnect backpressure_mode backpressure_timeout backpressure_alignment
        backpressure_deadline reconnect_shutdown pose_cancel config_duration_bounds)
    add_test(NAME gimbal.${scenario} COMMAND gimbal_lifecycle_tests ${scenario})
    set_tests_properties(gimbal.${scenario} PROPERTIES TIMEOUT 10)
endforeach()
