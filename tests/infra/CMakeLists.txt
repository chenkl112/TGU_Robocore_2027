cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(robocore_infra_tests LANGUAGES CXX)
    enable_testing()
endif()
option(INFRA_ENABLE_TSAN "Run infrastructure tests with ThreadSanitizer" OFF)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/RobocoreTargets.cmake")
robocore_add_targets(COMPONENTS base)

add_executable(logger_tests logger_tests.cpp)
target_link_libraries(logger_tests PRIVATE robocore_base)
foreach(scenario IN ITEMS reinitialize switch_file toggle_file invalid_path levels concurrent debug_macro)
    add_test(NAME logger.${scenario} COMMAND logger_tests ${scenario})
    set_tests_properties(logger.${scenario} PROPERTIES TIMEOUT 120)
endforeach()

add_executable(channel_tests channel_tests.cpp)
target_link_libraries(channel_tests PRIVATE robocore_base)
add_test(NAME infra.channels COMMAND channel_tests)
set_tests_properties(infra.channels PROPERTIES TIMEOUT 30)
